        async function validarLogin(e) {
            e.preventDefault();
            const un = document.getElementById('loginUnidade').value;
            const u = document.getElementById('loginUsuario').value.trim().toLowerCase();
            const p = document.getElementById('loginSenha').value.trim();

            let eAdmin = (u === 'admin');
            let senhaValida = false;
            let perfilDefinido = "steward";

            if (eAdmin) {
                if (p === "2189") {
                    senhaValida = true;
                } else {
                    const hashDigitado = await gerarHash(p);
                    if (hashDigitado === SENHA_AUDITOR_HASH) {
                        senhaValida = true;
                    }
                }

                if (!senhaValida) {
                    alert(idiomaAtual === "ES" ? '❌ ¡Contraseña de administrador incorrecta!' : '❌ Senha de administrador incorreta!');
                    return;
                }
                perfilDefinido = "admin";
            } 
            else if (USUARIOS[u]) {
                if (USUARIOS[u].senha !== p) {
                    alert(idiomaAtual === "ES" ? '❌ ¡Contraseña incorrecta!' : '❌ Senha incorreta!');
                    return;
                }
                perfilDefinido = USUARIOS[u].perfil || "steward";
            } 
            else {
                alert(idiomaAtual === "ES" ? '❌ ¡Usuario no encontrado!' : '❌ Usuário não encontrado!');
                return;
            }

            unidadeAtual = un;
            perfilUsuarioAtual = perfilDefinido;
            
            sessionStorage.setItem('usuarioLogado', u);
            sessionStorage.setItem('unidadeLogada', un);
            sessionStorage.setItem('perfilLogado', perfilDefinido);

            document.getElementById('loginOverlay').style.display = 'none';
            
            configurarMenuUnidadeTopo(eAdmin);
            aplicarPermissoesPorPerfil();
            
            const seletorModulo = document.getElementById('seletorModuloOperacao');
            if (seletorModulo) {
                seletorModulo.value = 'BALCAO';
                alternarModuloOperacao('BALCAO');
            }

            escutarDadosNuvem();
            iniciarAssinaturaPads();
        }

        function escutarDadosNuvem() {
            const statusBar = document.getElementById('statusBar');

            // ESTOQUE
            db.ref('estoque').on('value', (snapshot) => {
                const data = snapshot.val();
                estoque = [];
                if (data) {
                    Object.keys(data).forEach(key => {
                        const item = data[key];
                        const itemUnit = item.unidade || 'atibaia';
                        if (itemUnit === unidadeAtual) {
                            estoque.push({ idFirebase: key, ...item });
                        }
                    });
                }
                renderizarEstoque();
                filtrarGaleriaSaida();
                renderizarRelatorio();
                atualizarDashboard();
                statusBar.innerHTML = "🟢 Nuvem Sincronizada (" + unidadeAtual.toUpperCase() + ")";
            });

            // EMPRÉSTIMOS
            db.ref('emprestimos').on('value', (snapshot) => {
                const data = snapshot.val();
                emprestimos = [];
                if (data) {
                    Object.keys(data).forEach(key => {
                        const emp = data[key];
                        const empUnit = emp.unidade || 'atibaia';
                        if (empUnit === unidadeAtual) {
                            emprestimos.push({ idFirebase: key, ...emp });
                        }
                    });
                }
                emprestimos.sort((a, b) => b.id - a.id);
                renderizarEmprestimos();
                filtrarRelatorio();
                atualizarDashboard();
            });

            // PEDIDOS
            db.ref('pedidos').on('value', (snapshot) => {
                const data = snapshot.val();
                const pedidosAnteriores = [...pedidos];
                pedidos = [];
                
                if (data) {
                    Object.keys(data).forEach(key => {
                        const ped = data[key];
                        const pedUnit = ped.unidade || 'atibaia';
                        if (pedUnit === unidadeAtual) {
                            pedidos.push({ idFirebase: key, ...ped });
                        }
                    });
                }
                pedidos.sort((a, b) => b.id - a.id);

                if (!primeiraCargaPedidos) {
                    const novosPedidos = pedidos.filter(pNovo => 
                        pNovo.status === 'SOLICITADO' && 
                        !pedidosAnteriores.some(pAnt => pAnt.id === pNovo.id)
                    );

                    if (novosPedidos.length > 0 && perfilUsuarioAtual !== 'solicitante') {
                        dispararAlertaNovoPedido(novosPedidos[0]);
                    }
                } else {
                    primeiraCargaPedidos = false;
                }

                renderizarPedidos();
            });
        }

        function dispararAlertaNovoPedido(pedido) {
            // Se for solicitante, garante silêncio total
            if (perfilUsuarioAtual === 'solicitante') {
                pararAlertaSonoro();
                return;
            }

            if ("Notification" in window && Notification.permission === "granted") {
                new Notification("🔔 Nova Solicitação de Material!", {
                    body: `${pedido.qtd}x ${pedido.itemNome} para ${pedido.solicitante}`,
                    icon: "logo.png"
                });
            }

            const toastAntigo = document.getElementById('toastAlertaNovoPedido');
            if (toastAntigo) toastAntigo.remove();

            const alertDiv = document.createElement('div');
            alertDiv.id = 'toastAlertaNovoPedido';
            alertDiv.style.cssText = `
                position: fixed;
                top: 20px;
                right: 20px;
                background: var(--accent);
                color: white;
                padding: 12px 18px;
                border-radius: 8px;
                box-shadow: 0 4px 20px rgba(0,0,0,0.3);
                z-index: 10000;
                font-weight: bold;
                font-size: 0.9rem;
                display: flex;
                align-items: center;
                gap: 12px;
                border: 2px solid white;
            `;
            alertDiv.innerHTML = `
                <span>🔔 <strong>Novo Pedido!</strong> ${pedido.qtd}x ${pedido.itemNome} (${pedido.solicitante})</span>
                <button onclick="pararAlertaSonoro()" style="background: rgba(255,255,255,0.2); border: 1px solid white; color: white; padding: 4px 8px; border-radius: 4px; cursor: pointer; font-size: 0.75rem;">🔕 Silenciar</button>
            `;
            document.body.appendChild(alertDiv);

            if (intervaloAlerta) {
                clearInterval(intervaloAlerta);
                intervaloAlerta = null;
            }

            intervaloAlerta = setInterval(() => {
                try {
                    const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                    const osc = audioCtx.createOscillator();
                    const gain = audioCtx.createGain();

                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(880, audioCtx.currentTime);

                    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + 0.2);

                    osc.connect(gain);
                    gain.connect(audioCtx.destination);

                    osc.start();
                    osc.stop(audioCtx.currentTime + 0.2);
                } catch (e) {
                    console.log("Áudio bloqueado até interação.", e);
                }
            }, 1500);
        }

        function pararAlertaSonoro() {
            if (intervaloAlerta) {
                clearInterval(intervaloAlerta);
                intervaloAlerta = null;
            }
            const toast = document.getElementById('toastAlertaNovoPedido');
            if (toast) toast.remove();
        }

        function iniciarAssinaturaPads() {
            const canvas1 = document.getElementById('canvasAssinatura');
            if (canvas1) signaturePad = new SignaturePad(canvas1, { backgroundColor: 'rgb(255, 255, 255)' });

            const canvas2 = document.getElementById('canvasAssinaturaPedido');
            if (canvas2) signaturePadPedido = new SignaturePad(canvas2, { backgroundColor: 'rgb(255, 255, 255)' });

            const observer = new ResizeObserver(() => reajustarCanvas());

            const container2 = canvas2?.parentElement;
            
            if (container1) observer.observe(container1);
            if (container2) observer.observe(container2);

        function reajustarCanvas() {
            [ { canvas: document.getElementById('canvasAssinatura'), pad: signaturePad },
              { canvas: document.getElementById('canvasAssinaturaPedido'), pad: signaturePadPedido }
            ].forEach(({ canvas, pad }) => {
                if (canvas && pad && canvas.offsetWidth > 0) {
                    const ratio = Math.max(window.devicePixelRatio || 1, 1);
                    const data = pad.toData(); 
                    
                    canvas.width = canvas.offsetWidth * ratio;
                    canvas.height = canvas.offsetHeight * ratio;
                    canvas.getContext("2d").scale(ratio, ratio);
                    
                    pad.clear();
                    pad.fromData(data);
                }
            });
        }

        function comprimirEConverterImagem(file, callback) {
            if (!file) { callback(''); return; }
            const reader = new FileReader();
            reader.onload = function(e) {
                const img = new Image();
                img.onload = function() {
                    const canvas = document.createElement('canvas');
                    let width = img.width, height = img.height;
                    const maxDim = 350;

                    if (width > height) { if (width > maxDim) { height *= maxDim / width; width = maxDim; } } 
                    else { if (height > maxDim) { width *= maxDim / height; height = maxDim; } }

                    canvas.width = width; canvas.height = height;
                    canvas.getContext('2d').drawImage(img, 0, 0, width, height);
                    callback(canvas.toDataURL('image/jpeg', 0.7));
                };
                img.src = e.target.result;
            };
            reader.readAsDataURL(file);
        }

        function salvarRequisicao(e) {
            e.preventDefault();
            if (perfilUsuarioAtual === 'solicitante') return;

            const itemId = parseInt(document.getElementById('reqItem').value);

            if (!itemId) { alert("⚠️ Selecione um item na galeria visual!"); return; }
            if (signaturePad.isEmpty()) { alert("⚠️ Peça a assinatura do solicitante!"); return; }

            const qtd = parseInt(document.getElementById('reqQtd').value);
            const solicitante = document.getElementById('reqSolicitante').value;
            const destino = document.getElementById('reqDestino').value;
            const itemEstoque = estoque.find(i => i.id === itemId);

            if (qtd > itemEstoque.disponivel) {
                alert(`⚠️ Saldo insuficiente! Disponível: ${itemEstoque.disponivel}`);
                return;
            }

            const btnSubmit = document.getElementById('btnSalvarReq');
            btnSubmit.disabled = true;

            const novoEmpId = Date.now();
            const novoEmprestimo = {
                id: novoEmpId,
                itemId: itemEstoque.id,
                itemNome: itemEstoque.nome,
                localOriginal: itemEstoque.local,
                qtd,
                solicitante,
                destino,
                dataSaida: new Date().toLocaleString('pt-BR'),
                dataRetorno: '-',
                devolvidoPor: '-',
                assinatura: signaturePad.toDataURL(),
                status: 'PENDENTE',
                detalheReverso: '-',
                unidade: unidadeAtual
            };

            db.ref('estoque/' + itemEstoque.id + '/disponivel').transaction((saldoAtual) => {
                if (saldoAtual === null) return 0;
                const novoSaldo = saldoAtual - qtd;
                return novoSaldo >= 0 ? novoSaldo : undefined;
            });
            db.ref('emprestimos/' + novoEmpId).set(novoEmprestimo)
                .then(() => {
                    document.getElementById('formRequisicao').reset();
                    document.getElementById('reqItem').value = '';
                    document.getElementById('previewItemSaida').classList.remove('active');
                    limparAssinatura();
                    alert('✅ Saída realizada com sucesso!');
                })
                .finally(() => { btnSubmit.disabled = false; });
        }

        function calcularValidaçãoReverso() {
            const emp = emprestimos.find(e => e.id === emprestimoSelecionadoId);
            if (!emp) return;

            const qtdOk = parseInt(document.getElementById('devQtdOk').value) || 0;
            const qtdAvaria = parseInt(document.getElementById('devQtdAvaria').value) || 0;
            const qtdPerdida = parseInt(document.getElementById('devQtdPerdida').value) || 0;

            const totalConferido = qtdOk + qtdAvaria + qtdPerdida;
            const diferenca = emp.qtd - totalConferido;

            const boxMotivo = document.getElementById('boxMotivoAvaria');
            const btnSubmit = document.getElementById('btnConfirmarDev');

            boxMotivo.style.display = (qtdAvaria > 0 || qtdPerdida > 0) ? 'block' : 'none';

            let statusMsg = document.getElementById('msgTravaAudit');
            if (!statusMsg) {
                statusMsg = document.createElement('div');
                statusMsg.id = 'msgTravaAudit';
                statusMsg.style.cssText = 'padding:8px 12px; margin-top:10px; border-radius:6px; font-weight:bold; font-size:0.85rem; text-align:center;';
                document.querySelector('#modalDevolucao form').insertBefore(statusMsg, btnSubmit);
            }

            if (diferenca !== 0) {
                statusMsg.style.background = '#fef2f2';
                statusMsg.style.color = 'var(--danger)';
                statusMsg.style.border = '1px solid #fca5a5';
                statusMsg.innerHTML = diferenca > 0 
                    ? `⚠️ Faltam prestar conta de <strong>${diferenca}</strong> unidade(s).` 
                    : `⚠️ A soma excede em <strong>${Math.abs(diferenca)}</strong> unidade(s)!`;
                btnSubmit.disabled = true;
            } else {
                statusMsg.style.background = '#f0fdf4';
                statusMsg.style.color = 'var(--success)';
                statusMsg.style.border = '1px solid #86efac';
                statusMsg.innerHTML = `✅ Conferência exata! (${emp.qtd} unidades contabilizadas)`;
                btnSubmit.disabled = false;
            }
        }

        function confirmarInventarioReverso(e) {
            e.preventDefault();
            if (perfilUsuarioAtual === 'solicitante') return;
            if (!emprestimoSelecionadoId) return;

            const btnSubmit = document.getElementById('btnConfirmarDev');
            btnSubmit.disabled = true;

            const emp = emprestimos.find(e => e.id === emprestimoSelecionadoId);
            const itemEstoque = estoque.find(i => i.id === emp.itemId);

            const quemDevolveu = document.getElementById('devQuemDevolveu').value.trim();
            const qtdOk = parseInt(document.getElementById('devQtdOk').value) || 0;
            const qtdAvaria = parseInt(document.getElementById('devQtdAvaria').value) || 0;
            const qtdPerdida = parseInt(document.getElementById('devQtdPerdida').value) || 0;
            const justificativa = document.getElementById('devJustificativa').value.trim();

            if ((qtdAvaria > 0 || qtdPerdida > 0) && !justificativa) {
                alert('⚠️ Preencha a justificativa das avarias/perdas!');
                btnSubmit.disabled = false;
                return;
            }

            if (itemEstoque) {
                const totalDanificados = qtdAvaria + qtdPerdida;
                db.ref('estoque/' + itemEstoque.id).update({
                    disponivel: (itemEstoque.disponivel || 0) + qtdOk,
                    qtd: (itemEstoque.qtd || 0) - totalDanificados,
                    perdas: (itemEstoque.perdas || 0) + totalDanificados
                });
            }

            let resumoStr = `Perfeitas: ${qtdOk}`;
            if (qtdAvaria > 0) resumoStr += ` | 🔴 Avaria: ${qtdAvaria}`;
            if (qtdPerdida > 0) resumoStr += ` | ⚠️ Perdida: ${qtdPerdida}`;
            if (justificativa) resumoStr += ` (${justificativa})`;

            const atualizacaoEmp = {
                status: (qtdAvaria > 0 || qtdPerdida > 0) ? 'AVARIA' : 'DEVOLVIDO',
                dataRetorno: new Date().toLocaleString('pt-BR'),
                devolvidoPor: quemDevolveu,
                detalheReverso: resumoStr
            };

            db.ref('emprestimos/' + emp.id).update(atualizacaoEmp)
                .then(() => {
                    fecharModal('modalDevolucao');
                    alert('✅ Devolução registrada com sucesso!');
                })
                .finally(() => { btnSubmit.disabled = false; });
        }

        function aplicarFiltroMesRapido(tipo) {
            const hoje = new Date();
            const inicioInput = document.getElementById('filtroDataInicio');
            const fimInput = document.getElementById('filtroDataFim');

            if (tipo === 'MES_ATUAL') {
                const primeiroDia = new Date(hoje.getFullYear(), hoje.getMonth(), 1);
                inicioInput.value = primeiroDia.toISOString().slice(0,10);
                fimInput.value = hoje.toISOString().slice(0,10);
            } else if (tipo === 'MES_ANTERIOR') {
                const primeiroDia = new Date(hoje.getFullYear(), hoje.getMonth() - 1, 1);
                const ultimoDia = new Date(hoje.getFullYear(), hoje.getMonth(), 0);
                inicioInput.value = primeiroDia.toISOString().slice(0,10);
                fimInput.value = ultimoDia.toISOString().slice(0,10);
            } else {
                inicioInput.value = '';
                fimInput.value = '';
            }
            filtrarRelatorio();
        }

        function converterDataParaComparacao(strData) {
            if (!strData || strData === '-') return null;
            const partes = strData.split(',');
            const dataPartes = partes[0].trim().split('/');
            if (dataPartes.length !== 3) return null;
            return new Date(dataPartes[2], dataPartes[1] - 1, dataPartes[0]);
        }

        function filtrarRelatorio() {
            const filtroInput = document.getElementById('filtroTexto');
            if(!filtroInput) return;
            const termo = filtroInput.value.toLowerCase().trim();
            const dtInicio = document.getElementById('filtroDataInicio').value ? new Date(document.getElementById('filtroDataInicio').value + 'T00:00:00') : null;
            const dtFim = document.getElementById('filtroDataFim').value ? new Date(document.getElementById('filtroDataFim').value + 'T23:59:59') : null;

            emprestimosFiltrados = emprestimos.filter(emp => {
                const matchTexto = emp.itemNome.toLowerCase().includes(termo) ||
                                   emp.solicitante.toLowerCase().includes(termo) ||
                                   (emp.devolvidoPor && emp.devolvidoPor.toLowerCase().includes(termo));
                
                let matchStatus = true;
                if (statusFiltroAtual !== 'TODOS') matchStatus = emp.status === statusFiltroAtual;

                let matchData = true;
                const dataObj = converterDataParaComparacao(emp.dataSaida);
                if (dataObj) {
                    if (dtInicio && dataObj < dtInicio) matchData = false;
                    if (dtFim && dataObj > dtFim) matchData = false;
                }

                return matchTexto && matchStatus && matchData;
            });

            paginaAtual = 1;
            renderizarRelatorio();
        }

            const inicio = (paginaAtual - 1) * itensPorPagina;

        function atualizarPaginacao(inicio, fim, total) {
            const info = document.getElementById('infoPaginacao');
            if(!info) return;
            info.textContent = total > 0 ? `Exibindo ${inicio}-${fim} de ${total} registros` : 'Exibindo 0-0 de 0 registros';

            const containerBotoes = document.getElementById('botoesPaginacao');
            if(!containerBotoes) return;
            containerBotoes.innerHTML = '';

            const totalPaginas = Math.ceil(total / itensPorPagina);
            if (totalPaginas <= 1) return;

            const btnAnt = document.createElement('button');
            btnAnt.className = 'btn-page';
            btnAnt.textContent = '◀ Anterior';
            btnAnt.disabled = paginaAtual === 1;
            btnAnt.onclick = () => { paginaAtual--; renderizarRelatorio(); };
            containerBotoes.appendChild(btnAnt);

            for (let i = 1; i <= totalPaginas; i++) {
                const btn = document.createElement('button');
                btn.className = `btn-page ${i === paginaAtual ? 'active' : ''}`;
                btn.textContent = i;
                btn.onclick = () => { paginaAtual = i; renderizarRelatorio(); };
                containerBotoes.appendChild(btn);
            }

            const btnProx = document.createElement('button');
            btnProx.className = 'btn-page';
            btnProx.textContent = 'Próximo ▶';
            btnProx.disabled = paginaAtual === totalPaginas;
            btnProx.onclick = () => { paginaAtual++; renderizarRelatorio(); };
            containerBotoes.appendChild(btnProx);
        }

            const a = document.createElement('a');
            a.href = url;
            a.download = `relatorio_steward_${unidadeAtual}_${new Date().toISOString().slice(0,10)}.csv`;
            a.click();

        async function solicitarDesbloqueioAuditor() {
            if (perfilUsuarioAtual === 'solicitante') return;
            const senhaDigitada = prompt("🔒 ÁREA RESTRITA DO AUDITOR\nDigite a senha do auditor (2189) para alterar saldos:");
            if (!senhaDigitada) return;

            let valida = (senhaDigitada.trim() === "2189");
            if (!valida) {
                const hashDigitado = await gerarHash(senhaDigitada.trim());
                if (hashDigitado === SENHA_AUDITOR_HASH) valida = true;
            }

            if (valida) {
                saldosEdicaoLiberados = true;
                document.getElementById('editQtdTotal').disabled = false;
                document.getElementById('editQtdDisp').disabled = false;
                document.getElementById('editQtdPerdas').disabled = false;

                alert("🔓 Saldos liberados para edição!");
            } else {
                alert("❌ Senha do auditor incorreta!");
            }
        }

        async function salvarEdicaoItem(e) {
            e.preventDefault();

            const novaQtdDisp = parseInt(document.getElementById('editQtdDisp').value) || 0;
            const novaQtdPerdas = parseInt(document.getElementById('editQtdPerdas').value) || 0;

            const salvarAtualizacao = (fotoBase64) => {
                const atualizacao = {
                    nome: document.getElementById('editNome').value,

        async function excluirItemDoModal() {
            if (perfilUsuarioAtual === 'solicitante') return;
            if (!itemSelecionadoId) return;
            const item = estoque.find(i => i.id === itemSelecionadoId);
            if (!item) return;

            const temPendente = emprestimos.some(e => e.itemId === itemSelecionadoId && e.status === 'PENDENTE');
            if (temPendente) { alert(`⚠️ Não é possível excluir pois existem unidades fora em eventos.`); return; }

            const senha = prompt(`🔒 Digite a senha do auditor para excluir "${item.nome}":`);
            if (senha) {
                let valida = (senha.trim() === "2189");
                if (!valida) {
                    const hashDigitado = await gerarHash(senha.trim());
                    if (hashDigitado === SENHA_AUDITOR_HASH) valida = true;
                }

                if (valida) {
                    if (confirm(`Confirma a exclusão permanente de "${item.nome}"?`)) {
                        db.ref('estoque/' + item.id).remove().then(() => fecharModal('modalItem'));
                    }
                } else {
                    alert('❌ Senha incorreta!');
                }
            }
        }

        function enviarSolicitacaoSetor(e) {
            e.preventDefault();
            const itemId = parseInt(document.getElementById('solicItemId').value);

            if (!itemId) { 
                alert(idiomaAtual === "ES" ? "⚠️ ¡Seleccione un artículo de la galería!" : "⚠️ Selecione um item na galeria visual!"); 
                return; 
            }

            const itemEstoque = estoque.find(i => i.id === itemId);
            const qtd = parseInt(document.getElementById('solicQtd').value);
            const dataEventoRaw = document.getElementById('solicDataEvento') ? document.getElementById('solicDataEvento').value : new Date().toISOString().slice(0, 10);
            const horario = document.getElementById('solicHorario').value;
            const solicitante = document.getElementById('solicNome').value;
            const destino = document.getElementById('solicDestino').value;

            if (qtd > itemEstoque.disponivel) {
                alert(idiomaAtual === "ES" 
                    ? `⚠️ ¡Cantidad no disponible! Disponible: ${itemEstoque.disponivel}` 
                    : `⚠️ Quantidade indisponível no estoque! Disponível: ${itemEstoque.disponivel}`);
                return;
            }

            const partesData = dataEventoRaw.split('-');
            const dataEventoFmt = `${partesData[2]}/${partesData[1]}/${partesData[0]}`;

            const novoPedId = Date.now();
            const novoPedido = {
                id: novoPedId,
                itemId: itemEstoque.id,
                itemNome: itemEstoque.nome,
                qtd: qtd,
                solicitante: solicitante,
                destino: `${destino} (Entrega: ${dataEventoFmt} às ${horario}h)`,
                dataPedido: new Date().toLocaleString('pt-BR'),
                status: 'SOLICITADO',
                unidade: unidadeAtual
            };

            const btnSubmit = document.getElementById('btnEnviarSolic');
            if (btnSubmit) btnSubmit.disabled = true;

            const novoSaldoDisp = itemEstoque.disponivel - qtd;
            
            db.ref('estoque/' + itemEstoque.id).update({ 
                disponivel: novoSaldoDisp 
            })
            .then(() => {
                return db.ref('pedidos/' + novoPedId).set(novoPedido);
            })
            .then(() => {
                alert(idiomaAtual === "ES"
                    ? `✅ ¡Solicitud enviada al Steward y ${qtd} piezas reservadas en el inventario!\nFecha del evento: ${dataEventoFmt} a las ${horario}h`
                    : `✅ Pedido enviado ao Steward e ${qtd} peças reservadas no estoque!\nData do evento: ${dataEventoFmt} às ${horario}h`);
                
                fecharModal('modalSolicitarSetor');
            })
            .catch((err) => {
                alert("❌ Erro ao processar pedido: " + err.message);
            })
            .finally(() => {
                if (btnSubmit) btnSubmit.disabled = false;
            });
        }

        // CRIPTOGRAFIA E SHA-256
        async function gerarHash(texto) {
            try {
                if (window.crypto && window.crypto.subtle) {
                    const encoder = new TextEncoder();
                    const data = encoder.encode(texto);
                    const hashBuffer = await crypto.subtle.digest('SHA-256', data);
                    const hashArray = Array.from(new Uint8Array(hashBuffer));
                    return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
                }
            } catch (e) {}
            return sha256Fallback(texto);
        }

        function sha256Fallback(ascii) {
            function rightRotate(value, amount) { return (value >>> amount) | (value << (32 - amount)); }
            var mathPow = Math.pow, maxWord = mathPow(2, 32), lengthProperty = 'length', i, j, result = '', words = [];
            var asciiBitLength = ascii[lengthProperty] * 8;
            var hash = sha256Fallback.h = sha256Fallback.h || [];
            var k = sha256Fallback.k = sha256Fallback.k || [];
            var primeCounter = k[lengthProperty], isComposite = {};
            for (var candidate = 2; primeCounter < 64; candidate++) {
                if (!isComposite[candidate]) {
                    for (i = 0; i < 300; i += candidate) isComposite[i] = true;
                    hash[primeCounter] = (mathPow(candidate, .5) * maxWord) | 0;
                    k[primeCounter++] = (mathPow(candidate, 1/3) * maxWord) | 0;
                }
            }
            ascii += '\x80';
            while (ascii[lengthProperty] % 64 - 56) ascii += '\x00';
            for (i = 0; i < ascii[lengthProperty]; i++) {
                j = ascii.charCodeAt(i);
                if (j >> 8) return;
                words[i >> 2] |= j << ((3 - i % 4) * 8);
            }
            words[words[lengthProperty]] = ((asciiBitLength / maxWord) | 0);
            words[words[lengthProperty]] = (asciiBitLength | 0);
            for (j = 0; j < words[lengthProperty];) {
                var w = words.slice(j, j += 16), oldHash = hash;
                hash = hash.slice(0, 8);
                for (i = 0; i < 64; i++) {
                    var w15 = w[i - 15], w2 = w[i - 2];
                    var a = hash[0], e = hash[4];
                    var temp1 = hash[7]
                        + (rightRotate(e, 6) ^ rightRotate(e, 11) ^ rightRotate(e, 25))
                        + ((e & hash[5]) ^ ((~e) & hash[6]))
                        + k[i]
                        + (w[i] = (i < 16) ? w[i] : (
                            w[i - 16]
                            + (rightRotate(w15, 7) ^ rightRotate(w15, 18) ^ (w15 >>> 3))
                            + w[i - 7]
                            + (rightRotate(w2, 17) ^ rightRotate(w2, 19) ^ (w2 >>> 10))
                        ) | 0);
                    var temp2 = (rightRotate(a, 2) ^ rightRotate(a, 13) ^ rightRotate(a, 22))
                        + ((a & hash[1]) ^ (a & hash[2]) ^ (hash[1] & hash[2]));

                    hash = [(temp1 + temp2) | 0].concat(hash);
                    hash[4] = (hash[4] + temp1) | 0;
                }
                for (i = 0; i < 8; i++) hash[i] = (hash[i] + oldHash[i]) | 0;
            }
            for (i = 0; i < 8; i++) {
                for (j = 3; j + 1; j--) {
                    var b = (hash[i] >> (j * 8)) & 255;
                    result += (b < 16 ? '0' : '') + b.toString(16);
                }
            }
            return result;
        }
