const SENHA_PADRAO = "bourbon123";

function autenticarUsuario(event) {
    event.preventDefault();
    const usuarioInput = document.getElementById('loginUsuario').value.trim();
    const senhaInput = document.getElementById('loginSenha').value.trim();
    const msgErro = document.getElementById('loginErro');

    // Validação de credenciais
    if (usuarioInput === USUARIO_PADRAO && senhaInput === SENHA_PADRAO) {
        msgErro.style.display = 'none';
        
        // Salva a sessão no navegador
        sessionStorage.setItem('usuarioLogado', usuarioInput);
        
        // Oculta a tela de login
        document.getElementById('loginOverlay').style.display = 'none';
        document.getElementById('formLogin').reset();
    } else {
        msgErro.style.display = 'block';
    }
}

function verificarSessaoLogin() {
    const usuarioLogado = sessionStorage.getItem('usuarioLogado');
    const overlay = document.getElementById('loginOverlay');

    if (usuarioLogado) {
        overlay.style.display = 'none';
    } else {
        overlay.style.display = 'flex';
    }
}

function fazerLogout() {
    if (confirm("Deseja realmente sair do sistema?")) {
        sessionStorage.removeItem('usuarioLogado');
        document.getElementById('loginOverlay').style.display = 'flex';
    }
}

        function inicializarCanvasAssinatura() {
            canvasAssinatura = document.getElementById('canvasAssinatura');

            canvasAssinatura.width = canvasAssinatura.offsetWidth || 400;

        async function sha256(str) {
            const buf = await crypto.subtle.digest("SHA-256", new TextEncoder("UTF-8").encode(str));
            return Array.prototype.map.call(new Uint8Array(buf), x => (('00' + x.toString(16)).slice(-2))).join('');
        }

        function confirmarDevolucaoAB(event) {
            event.preventDefault();

            const emUsoId = document.getElementById('devEmUsoId').value;
            const itemId = document.getElementById('devItemId').value;
            const qtdEmUsoAtual = parseInt(document.getElementById('devQtdEmUso').value, 10);
            const qtdDevolvida = parseInt(document.getElementById('devQtdDevolvida').value, 10);

            if (isNaN(qtdDevolvida) || qtdDevolvida <= 0 || qtdDevolvida > qtdEmUsoAtual) {
                return alert('Quantidade devolvida inválida.');
            }

            const itemEstoque = listaEstoque.find(i => i.id === itemId);
            if (!itemEstoque) return alert('Erro ao localizar o item no estoque.');

            const novaQtdDisponivel = itemEstoque.qtdDisponivel + qtdDevolvida;
            const dataHoraAtual = new Date().toLocaleString('pt-BR');

            if (firebaseAtivo) {
                refEstoque.child(itemId).update({ qtdDisponivel: novaQtdDisponivel })
                    .then(() => {
                        if (qtdDevolvida === qtdEmUsoAtual) return refEmUso.child(emUsoId).remove();
                        else return refEmUso.child(emUsoId).update({ qtdEmUso: qtdEmUsoAtual - qtdDevolvida });
                    })
                    .then(() => {
                        return refMovimentacoes.push().set({
                            itemId: itemId, itemNome: itemEstoque.nome, categoria: itemEstoque.categoria,
                            tipoSaida: '↩️ Devolução ao Estoque', qtd: qtdDevolvida, responsavel: 'Retorno de Empréstimo',
                            motivo: 'Devolução realizada no Balcão', dataHora: dataHoraAtual
                        });
                    })
                    .then(() => {
                        finalizarDevolucao();
                    })
                    .catch(err => alert('Erro Firebase: ' + err.message));
            } else {
                itemEstoque.qtdDisponivel = novaQtdDisponivel;
                localStorage.setItem('estoque_ab', JSON.stringify(listaEstoque));

                if (qtdDevolvida === qtdEmUsoAtual) {
                    listaEmUso = listaEmUso.filter(u => u.id !== emUsoId);
                } else {
                    const indexUso = listaEmUso.findIndex(u => u.id === emUsoId);
                    if (indexUso !== -1) listaEmUso[indexUso].qtdEmUso = qtdEmUsoAtual - qtdDevolvida;
                }
                localStorage.setItem('em_uso_ab', JSON.stringify(listaEmUso));

                listaMovimentacoes.unshift({
                    id: 'mov_' + Date.now(), itemId: itemId, itemNome: itemEstoque.nome, categoria: itemEstoque.categoria,
                    tipoSaida: '↩️ Devolução ao Estoque', qtd: qtdDevolvida, responsavel: 'Retorno de Empréstimo',
                    motivo: 'Devolução realizada no Balcão', dataHora: dataHoraAtual
                });
                localStorage.setItem('movimentacoes_ab', JSON.stringify(listaMovimentacoes));

                finalizarDevolucao();
            }

            const itemId = document.getElementById('saidaItem').value;
            const tipoMovimento = document.getElementById('saidaTipoMovimento').value;
            const qtdSaida = parseInt(document.getElementById('saidaQtd').value, 10);
            const responsavel = document.getElementById('saidaResponsavel').value.trim();
            const motivo = document.getElementById('saidaMotivo').value.trim();
            const assinaturaBase64 = obterAssinaturaBase64();

            const item = listaEstoque.find(i => i.id === itemId);
            
            if (!item || isNaN(qtdSaida) || qtdSaida <= 0 || qtdSaida > item.qtdDisponivel) {
                alert('Quantidade solicitada é inválida ou maior do que o estoque disponível.');
                if (btn) { btn.disabled = false; btn.innerHTML = '✅ Confirmar Saída'; }
                return;
            }

            let novaQtdTotal = item.qtdTotal;
            if (tipoMovimento !== 'emprestimo') {
                novaQtdTotal = item.qtdTotal - qtdSaida;
            }
            
            const novaQtdDisponivel = item.qtdDisponivel - qtdSaida;
            const dataHoraAtual = new Date().toLocaleString('pt-BR');

            if (firebaseAtivo) {
                refEstoque.child(itemId).update({ 
                    qtdTotal: novaQtdTotal,
                    qtdDisponivel: novaQtdDisponivel 
                })
                    .then(() => {
                        if (tipoMovimento === 'emprestimo') {
                            return refEmUso.push().set({
                                itemId, itemNome: item.nome, categoria: item.categoria,
                                qtdEmUso: qtdSaida, responsavel, motivo, assinatura: assinaturaBase64, dataHora: dataHoraAtual
                            });
                        }
                    })
                    .then(() => {
                        return refMovimentacoes.push().set({
                            itemId, itemNome: item.nome, categoria: item.categoria,
                            tipoSaida: tipoMovimento === 'emprestimo' ? '🔄 Empréstimo' : '🔴 Saída Definitiva',
                            qtd: qtdSaida, responsavel, motivo, dataHora: dataHoraAtual
                        });
                    })
                    .then(() => {
                        finalizarSaida();
                    })
                    .catch(err => {
                        alert('Erro Firebase: ' + err.message);
                        if (btn) { btn.disabled = false; btn.innerHTML = '✅ Confirmar Saída'; }
                    });
            } else {
                item.qtdTotal = novaQtdTotal;
                item.qtdDisponivel = novaQtdDisponivel;
                localStorage.setItem('estoque_ab', JSON.stringify(listaEstoque));

                if (tipoMovimento === 'emprestimo') {
                    listaEmUso.push({
                        id: 'emuso_' + Date.now(), itemId, itemNome: item.nome, categoria: item.categoria,
                        qtdEmUso: qtdSaida, responsavel, motivo, assinatura: assinaturaBase64, dataHora: dataHoraAtual
                    });
                    localStorage.setItem('em_uso_ab', JSON.stringify(listaEmUso));
                }

                listaMovimentacoes.unshift({
                    id: 'mov_' + Date.now(), itemId, itemNome: item.nome, categoria: item.categoria,
                    tipoSaida: tipoMovimento === 'emprestimo' ? '🔄 Empréstimo' : '🔴 Saída Definitiva',
                    qtd: qtdSaida, responsavel, motivo, dataHora: dataHoraAtual
                });
                localStorage.setItem('movimentacoes_ab', JSON.stringify(listaMovimentacoes));

                finalizarSaida();
            }

            function finalizarSaida() {
                alert('Saída de "' + item.nome + '" registrada com sucesso!');
                document.getElementById('formSaidaBalcao').reset();
                limparAssinatura();
                renderizarTabelasEstoque();
                renderizarTabelaEmUso();
                renderizarTabelaHistorico();
                preencherSelectBalcao();
                if (btn) { btn.disabled = false; btn.innerHTML = '✅ Confirmar Saída'; }
            }
        }

        function carregarMovimentacoesAB() {
            if (firebaseAtivo) {
                refMovimentacoes.on('value', snapshot => {
                    listaMovimentacoes = [];
                    const data = snapshot.val();
                    if (data) Object.keys(data).forEach(key => listaMovimentacoes.push({ id: key, ...data[key] }));
                    listaMovimentacoes.reverse();
                    renderizarTabelaHistorico();
                });
            } else {
                const localData = localStorage.getItem('movimentacoes_ab');
                listaMovimentacoes = localData ? JSON.parse(localData) : [];
                renderizarTabelaHistorico();
            }
        }

        function renderizarTabelaHistorico() {
            const tbody = document.getElementById('tbHistoricoMovimentacoes');
            tbody.innerHTML = '';

            if (listaMovimentacoes.length === 0) {
                tbody.innerHTML = '<tr><td colspan="7" style="text-align:center; color:#777;">Nenhuma movimentação registrada.</td></tr>';
                return;
            }

            listaMovimentacoes.forEach(mov => {
                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td>${mov.dataHora}</td>
                    <td><strong>${mov.itemNome}</strong></td>
                    <td>${mov.tipoSaida}</td>
                    <td>${mov.categoria}</td>
                    <td>${mov.qtd}</td>
                    <td>${mov.responsavel}</td>
                    <td>${mov.motivo}</td>
                `;
                tbody.appendChild(tr);
            });
        }

        function exportarRelatorioCSVAB() {
            if (listaMovimentacoes.length === 0) return alert('Sem dados para exportar.');

            let csvContent = "data:text/csv;charset=utf-8,\uFEFF";
            csvContent += "Data/Hora;Item;Tipo de Saida;Categoria;Quantidade;Responsavel;Motivo/Destino\n";

            listaMovimentacoes.forEach(m => {
                csvContent += [`"${m.dataHora}"`, `"${m.itemNome}"`, `"${m.tipoSaida}"`, `"${m.categoria}"`, `"${m.qtd}"`, `"${m.responsavel}"`, `"${m.motivo}"`].join(";") + "\n";
            });

            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", `relatorio_estoque_AB_${new Date().toISOString().slice(0,10)}.csv`);
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }

        function renderizarGraficosAB() {
            if (typeof Chart === 'undefined') return;

            const categoriasCount = {};
            listaEstoque.forEach(item => {
                categoriasCount[item.categoria] = (categoriasCount[item.categoria] || 0) + item.qtdTotal;
            });

            const ctxCat = document.getElementById('chartCategorias');
            if (ctxCat) {
                if (chartCatInstance) chartCatInstance.destroy();
                chartCatInstance = new Chart(ctxCat, {
                    type: 'doughnut',
                    data: {
                        labels: Object.keys(categoriasCount),
                        datasets: [{
                            data: Object.values(categoriasCount),
                            backgroundColor: ['#00463F', '#10b981', '#f59e0b', '#ef4444', '#8b5cf6', '#ec4899', '#06b6d4']
                        }]
                    },
                    options: { responsive: true, plugins: { title: { display: true, text: 'Total por Categoria' } } }
                });
            }

            const tiposCount = { 'Mobília': 0, 'Rouparia': 0, 'Materiais A&B': 0 };
            listaEstoque.forEach(item => {
                if (item.tipo === 'mobilia') tiposCount['Mobília'] += item.qtdTotal;
                else if (item.tipo === 'rouparia') tiposCount['Rouparia'] += item.qtdTotal;
                else if (item.tipo === 'materiais') tiposCount['Materiais A&B'] += item.qtdTotal;
            });

            const ctxTipo = document.getElementById('chartTiposEstoque');
            if (ctxTipo) {
                if (chartTipoInstance) chartTipoInstance.destroy();
                chartTipoInstance = new Chart(ctxTipo, {
                    type: 'pie',
                    data: {
                        labels: Object.keys(tiposCount),
                        datasets: [{
                            data: Object.values(tiposCount),
                            backgroundColor: ['#00463F', '#10b981', '#f59e0b']
                        }]
                    },
                    options: { responsive: true, plugins: { title: { display: true, text: 'Tipos de Estoque' } } }
                });
            }
        }
